**background.js**:
- Polls http://127.0.0.1:8765/api/ext/poll every 2 seconds
- On response, parse JSON
- If there's an action, forward to active tab's content script via chrome.tabs.sendMessage
- Get the result back
- POST the result to http://127.0.0.1:8765/api/ext/result
- Handle fetch errors gracefully (try/catch)

```javascript
const POLL_URL = 'http://127.0.0.1:8765/api/ext/poll';
const RESULT_URL = 'http://127.0.0.1:8765/api/ext/result';
const POLL_INTERVAL = 2000;

async function poll() {
  try {
    const res = await fetch(POLL_URL);
    if (!res.ok) return;
    const data = await res.json();
    if (!data || !data.action) return;

    // Get active tab
    const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });
    if (!tab || !tab.id) return;

    // Forward to content script
    const result = await chrome.tabs.sendMessage(tab.id, {
      type: 'MANTRA_COMMAND',
      id: data.id,
      action: data.action,
      selector: data.selector,
      text: data.text
    });

    // Post result back
    try {
      await fetch(RESULT_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          id: data.id,
          result: result !== undefined ? result : { error: 'no response from content script' }
        })
      });
    } catch (e) {
      console.error('Mantra Bridge: failed to POST result:', e);
    }
  } catch (e) {
    // Network error, server down, etc. - don't crash
    console.error('Mantra Bridge: poll error:', e);
  }
}

// Start polling
setInterval(poll, POLL_INTERVAL);
poll(); // immediate first poll
```

**content.js**:
- Listens for messages of type 'MANTRA_COMMAND'
- Performs the action:
  - "click": find element by selector, click it, return result
  - "type": find element by selector, type text (set value, dispatch events), return result
  - "read": find element by selector, read text content or value, return result
- Returns the result

```javascript
chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
  if (msg.type !== 'MANTRA_COMMAND') return;

  const { id, action, selector, text } = msg;

  try {
    const el = document.querySelector(selector);
    if (!el) {
      sendResponse({ error: 'selector not found: ' + selector });
      return;
    }

    switch (action) {
      case 'click':
        el.click();
        sendResponse({ ok: true, action: 'click', selector });
        break;
      case 'type':
        el.focus();
        el.value = text || '';
        el.dispatchEvent(new Event('input', { bubbles: true }));
        el.dispatchEvent(new Event('change', { bubbles: true }));
        sendResponse({ ok: true, action: 'type', selector, value: el.value });
        break;
      case 'read':
        const value = el.value !== undefined ? el.value : el.textContent;
        sendResponse({ ok: true, action: 'read', selector, value: value.trim() });
        break;
      default:
        sendResponse({ error: 'unknown action: ' + action });
    }
  } catch (e) {
    sendResponse({ error: e.message });
  }

  return true; // keep channel open for async sendResponse - actually it's sync here so not needed, but safe
});
```
